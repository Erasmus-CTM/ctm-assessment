// Applets that size themselves once their pane is shown: 4 × 60 px.
document.querySelectorAll('.uke3-grow').forEach(block => {
  let grown = false;
  new ResizeObserver(() => {
    if (grown || !block.offsetParent) return;
    grown = true;
    let steps = 0;
    const timer = setInterval(() => {
      block.style.height = (40 + 60 * ++steps) + 'px';
      if (steps === 4) clearInterval(timer);
    }, 120);
  }).observe(block);
});
</script>
